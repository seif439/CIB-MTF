انسخ الأكواد التالية للربط بين سيرفر Node.js وروبلوكس استوديو

ضع هذا الكود في ملف index.js على Replit: 
const express = require('express');
const app = express();
app.use(express.json());

const PORT = process.env.PORT || 3000;
const SECRET_KEY = process.env.ROBLOX_SECRET || "my_secret_key_123";

let pendingCommands = [];
let livePlayersData = [];

// API Endpoint for Roblox Sync
app.all('/api/public/roblox/sync', (req, res) => {
    const authHeader = req.headers['authorization'];
    if (authHeader !== SECRET_KEY) return res.status(401).json({ error: 'Unauthorized' });

    if (req.method === 'POST' && req.body.players) {
        livePlayersData = req.body.players;
    }

    const cmds = [...pendingCommands];
    pendingCommands = [];
    res.json({ commands: cmds });
});

// Anti-Cheat Webhook Endpoint
app.post('/api/public/roblox/anticheat', (req, res) => {
    const authHeader = req.headers['authorization'];
    if (authHeader !== SECRET_KEY) return res.status(401).json({ error: 'Unauthorized' });
    
    console.log("AntiCheat Alert:", req.body);
    res.json({ status: 'received' });
});

app.listen(PORT, () => console.log(`RoAdmin Server Running on Port ${PORT}`));
ضع هذا الكود داخل ServerScriptService في روبلوكس: 
local HttpService = game:GetService("HttpService")
local Players = game:GetService("Players")

local BRIDGE_URL = "https://YOUR-APP-NAME.replit.app/api/public/roblox/sync"
local SECRET_KEY = "my_secret_key_123"

local function collectPlayers()
	local list = {}
	for _, p in ipairs(Players:GetPlayers()) do
		local char = p.Character
		local hrp = char and char:FindFirstChild("HumanoidRootPart")
		local pos = hrp and hrp.Position or Vector3.zero
		table.insert(list, {
			name = p.Name,
			displayName = p.DisplayName,
			rank = p:GetAttribute("Rank") or "لاعب",
			position = { x = math.round(pos.X), y = math.round(pos.Y), z = math.round(pos.Z) },
			tools = {}
		})
	end
	return list
end

task.spawn(function()
	while true do
		task.wait(3)
		pcall(function()
			local res = HttpService:RequestAsync({
				Url = BRIDGE_URL,
				Method = "POST",
				Headers = { ["Content-Type"] = "application/json", ["Authorization"] = SECRET_KEY },
				Body = HttpService:JSONEncode({ players = collectPlayers() })
			})
		end)
	end
end)
